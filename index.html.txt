        async function fetchFileList(path) {
            try {
                const res = await fetch(GITHUB_API_URL + path, { headers: { 'Authorization': `token ${GITHUB_TOKEN}` } });
                if(!res.ok) return [];
                const data = await res.json();
                return Array.isArray(data) ? data.filter(f => f.name.endsWith('.pdf')).map(f => ({ fullPath: `${path}/${f.name}`, name: f.name, download_url: f.download_url })) : [];

        function renderFileList() {

                return path.includes(catKey) && (typeKey ? path.includes(typeKey) : true);

            if (relevantFiles.length === 0) {
                container.innerHTML = '<div class="text-center text-xs text-slate-400 py-4">Belge bulunamadı.</div>';

        async function readFile(file) {
            if(file.type === "application/pdf") {
                const ab = await file.arrayBuffer();
                const pdf = await pdfjsLib.getDocument(ab).promise;
                let txt = "";
                const max = Math.min(pdf.numPages, 5); 
                for(let i=1; i<=max; i++) txt += (await (await pdf.getPage(i)).getTextContent()).items.map(s=>s.str).join(' ')+"\n";
                return txt;
            } return await file.text();
        }
